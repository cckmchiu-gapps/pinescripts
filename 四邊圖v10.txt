        r := float(bar_index) + (ts - float(time)) / 86400000.0 * barsPerDay
    else
        for i = 1 to 800
            if na(r) and not na(time[i]) and float(time[i]) <= ts
                // time[i] 係「較早（左）」，time[i-1] 係「較遲（右）」
                float tA = float(time[i])
                float tB = not na(time[i - 1]) ? float(time[i - 1]) : tA + avgBarMs
                float d = tB - tA
                float fr = d > 0 ? (ts - tA) / d : 0.0
                // 由距離 i 嗰根向右移 fr 根 → 棒索引【加】
                r := float(bar_index - i) + fr
    na(r) ? float(bar_index) : r

// K 棒索引（小數）→ 時間戳：用相鄰兩根 K 棒嘅實際時間線性內插
//   ★ 數學保證：TradingView 嘅「時間 → 像素」喺【相鄰兩根 K 棒之間】一定係線性
//     （假期空白槽只係令該段斜率高啲，仍然係一條直線段）
//     → 所以線性內插係【精確】嘅，唔需要任何吸附 fallback。
f_barToTs(float b) =>
    float k = float(bar_index) - b          // 距離當前棒幾多根（>0 過去、<0 未來）
    float r = na
    if k <= 0.0
        // ★ V10.0：同 f_tsToBar 用【同一個】日曆感知係數
        //   → round-trip 依然精確（兩個方向嘅誤差互相抵消）。
        r := float(time) + (float(b) - float(bar_index)) / barsPerDay * 86400000.0
    else
        int kf = int(math.floor(k))
        float fr = k - kf                    // [0, 1)
        if kf >= 799
            float tRef = not na(time[799]) ? float(time[799]) : float(time)
            r := tRef - (k - 799.0) * avgBarMs
        else
            // b 位於「距離 kf」同「距離 kf+1」兩根 K 棒之間
            float tA = not na(time[kf]) ? float(time[kf]) : float(time)
            float tB = not na(time[kf + 1]) ? float(time[kf + 1]) : tA - avgBarMs
            float d = math.max(tA - tB, 0.0)
            r := tA - fr * d
    na(r) ? float(time) : r

// ── ★ V10.0：x 解析器（輸入一律係【K 棒索引】）──
//   ① 棒索引模式 → 直接四捨五入成整數 K 棒索引（零轉換）
//      再加一層硬夾限 [bar_index − 9999, bar_index + 500]，
//      確保永遠唔會踩到 Pine 嘅 runtime error。
//   ② 時間模式   → 轉時間戳（v9.9 舊行為；冇 500 根限制）
//   xb = true 用棒索引、false 用時間。
f_xInt(float b, bool xb) =>
    xb ? int(math.min(math.max(math.round(b), float(bar_index) - 9999.0), float(bar_index) + 500.0)) : int(math.round(f_barToTs(b)))

// ── 算出「某年某月第 N 個星期 X」的 UNIX 毫秒時間戳 ──
//   ★ V10.0：一律用【12:00 UTC】做基準。
//     舊版用 00:00 UTC → 對 UTC−12 ~ UTC+11 以外嘅交易所（美股 16:00 ET 等）
//     會落到前一個日曆日 → 結算日偏一日。
//     12:00 UTC 對 UTC−12 ~ UTC+11 都係同一日曆日
//     （唯一例外：Pacific/Auckland 夏令時 UTC+13，已在說明文件註明）。
//   ★ 亦完全避開 timestamp() 嘅 timezone 參數 → 唔會踩 CE10123 陷阱。
f_settleTs(int yy, int mm, int dow, int occ) =>
    int tsFirst = timestamp(yy, mm, 1, 12, 0, 0)
    int fd = dayofweek(tsFirst)
    int delta = (dow - fd + 7) % 7
    int firstDay = 1 + delta
    int yyN = mm == 12 ? yy + 1 : yy
    int mmN = mm == 12 ? 1 : mm + 1
    int dim = dayofmonth(timestamp(yyN, mmN, 1, 12, 0, 0) - 86400000)
    int dayNum = firstDay + (occ - 1) * 7
    if dayNum > dim
        dayNum := firstDay + int(math.floor((dim - firstDay) / 7.0)) * 7
    timestamp(yy, mm, dayNum, 12, 0, 0)

    // 結算日參考線亦對齊到同一根 K 棒（避免同方框右邊差半根棒）
    // ★ V10.3：結算日參考線一律用【真實時間戳】（唔跟 xMode，因為佢係日曆日期）
    //   原本喺時間模式會 round 到整數棒 → 同方框右邊差 ≤0.5 棒；V10.3 移除 round。

    // ═══ ★ 手柄 3 · 左邊（左右拖）═══
    // ★ V10.4：同右邊一樣，改用【固定時間窗】判定（原本用可視範圍 → 會因滾動而反轉）
    bool leftOn = (float(hLeft) >= hValidLo) and (float(hLeft) <= hValidHi)
    float xAutoLeftBar = xRightABar - totalBars
    float xLeftBBar = leftOn ? (f_tsToBar(float(hLeft)) + offXb) : xAutoLeftBar

    // ★ v9.2 修正：兩條邊交叉都唔再回退寬度 —— 用 min/max，
    //   方框永遠由左至右，而兩條邊【依然各自貼實自己嘅手柄】。
    // ★ V10.0：xLeftBarRaw / xRightBarRaw 就係【左右兩邊嘅原始錨點】，
    //   下面直接各自 round → 兩邊都精確對齊手柄 / 結算日。
    float xLeftBarRaw = math.min(xLeftBBar, xRightABar)
    float xRightBarRaw = math.max(xLeftBBar, xRightABar)

    // ── ★ V10.0：X 軸量化（邊線優先）──
    //   ① 左邊、右邊【各自】round 到整數棒 → 兩邊都精確對齊手柄 / 結算日
    //   ② 內部分割點四捨五入 → 四段極差 ≤ 1 根棒（約 6px）
    //   ③ 想要「四段 0 誤差」→ ② 揀「分割優先（吸附 4 的倍數）」
    float xLeftBar = useBarIdx ? math.round(xLeftBarRaw) : xLeftBarRaw
    float xRightBar = useBarIdx ? math.round(xRightBarRaw) : xRightBarRaw
    if useBarIdx and snapQ4
        // 分割優先（可選）：以【左邊】為基準，格寬吸附 4 的倍數
        // → 四段 0 誤差，但右邊可能偏離錨點 ≤ 2 根棒
        float cw = math.max(math.round((xRightBar - xLeftBar) / cols), 4.0)
        cw := math.max(4.0 * math.round(cw / 4.0), 4.0)
        xRightBar := xLeftBar + cw * cols
    float boxWBars = math.max(xRightBar - xLeftBar, 1.0)
    float cellWBars = boxWBars / cols

    // ── ★ V10.0：xloc.bar_index 只可畫到 [bar_index − 9999, bar_index + 500] ──
    //   ★ 唔用「範圍平移」！平移會令【左右邊線偏離手柄 / 結算日】，
    //     即係親手破壞「右邊對齊結算日」嘅語義。
    //   改為：只要任一邊超出 → 直接降級去【時間模式】。
    //     xloc.bar_time 冇 500 根限制，而且用【真實時間戳】→ 邊線一定對齊。
    //   代價：水平分割會受假期影響。但「邊線對齊」遠比「分割 0 誤差」重要。
    bool xUseBar = useBarIdx
    bool xDegraded = false
    if useBarIdx
        float xMargin = boxWBars * 0.05          // 對位診斷線會伸出 5%
        if xLeftBar < float(bar_index) - 9999.0 + xMargin or xRightBar > float(bar_index) + 500.0 - xMargin
            xUseBar := false                     // ★ 降級：改用 xloc.bar_time
            xDegraded := true

    // ═══ ★ 手柄 2 · 底邊（上下拖）═══
    //   ★ v9.2 核心修正：手柄有設定 → 邊 = 手柄值，【零偏移】
    //     ⑤ 垂直位移（offY）只作用於【自動位置】，而且全程只加一次。
    //     咁樣就保證「手柄喺邊，方框條邊就喺邊」，唔會偏離滑鼠線。
    float refH = rows * baseH
    // ★ V10.10 精簡：冇咗「垂直基準」二選一 → 一律「貼近現價」居中。
    float yAutoBot = close - refH * 0.5
    // ★ V10.7：統一價格來源。
    //   · botSet / topSet = 手柄【原始值】有冇設定（=0 視為未設定）
    //   · botOn  / topOn  = 「有嘢話事」（均價模式永遠 true）
    //   ★ 注意：foreign 判定要用 botSet（唔可以用 botOn）——
    //     均價模式 botOn 永遠 true，用咗就會令「手柄=0」被誤判成「外來值」。
    bool botSet = hBot != 0.0
    bool topSet = hTop != 0.0
    bool botOn = useMA or botSet
    bool topOn = useMA or topSet

    // ★ V10.10 精簡：【各標的記憶表】已移除 → 方框來源只剩「手柄 / 均價起步」。
    //   ① 均價模式 → 方框 = 均價 ∓%（除非用戶開咗「★ 用手柄接手」→ 用手柄）
    //   ② 手柄模式 → 方框 = 手柄值（「自動適應」會先做 foreign 判定）
    //   ③ 兩者都未設定 → 自動位置（現價 ± 自動高度 %）

    // ═══ ★ v9.4 自動適應（★ V10.2 起：所有模式都生效，除「均價做起步」模式）═══
    //   TradingView 嘅手柄值係跟 Chart Layout 走、唔跟 Symbol 走。
    //   所以切換股票時，手柄值（絕對價格）會原封不動帶過去 →
    //   對新標的嚟講係唔合理嘅數字（例如 24800 對美股 $50）→ 方框飛走。
    //   ★ 只係「暫時唔理」嗰個值，【唔會改寫】佢 → 切返原股票即還原。
    //
    //   ★ v9.5 修正 v9.4 嘅誤觸發 bug：
    //     v9.4：adapted = botForeign or topForeign
    //       → 只要【任何一邊】遠，兩邊一齊失效。
    //       方框一寬（低邊離現價遠）就誤觸發 →【兩條邊都脫開手柄】。
    //     v9.5：兩邊都有設定 → 要【兩邊都遠】先觸發（切股票正正係咁）；
    //           只有一邊設定 → 嗰邊遠就觸發。
    float adaptLim = close * adaptPct / 100.0
    // ★ V10.2：唔再限制模式 —— 手柄值離現價太遠就自動避開。
    //   ★ 呢個就係「切換第二隻股票 → 方框完全唔見」嘅修復。
    // ★ V10.10：記憶表已移除 → adaptOn 唔再需要 `not slotHit`。
    bool adaptOn = autoAdapt and adaptPct > 0.0 and close > 0.0
    // ★ V10.7：均價模式【豁免】自動適應。
    //   若唔豁免：你一拖手柄拖遠少少 → 離現價 > 適應門檻 → 被判定「屬上隻股票」
    //   → 手柄即刻失效、方框彈返均價位 =「拖唔到」（正正就係用戶唔要嘅嘢）。
    //   ★ 所以均價模式：唔做 foreign 判定 → 拖幾遠都得。
    bool botForeign = adaptOn and (not useMA) and botSet and (math.abs(hBot - close) > adaptLim)
    bool topForeign = adaptOn and (not useMA) and topSet and (math.abs(hTop - close) > adaptLim)
    bool bothSet = botSet and topSet
    bool adapted = bothSet ? (botForeign and topForeign) : (botForeign or topForeign)

    // ═══ ★ V10.7/V10.8：有效手柄值 hBotE / hTopE —— 下游（畫框 / 診斷線 / 標籤）一律用呢兩個 ═══
    //   非均價模式：hBotE = hBot（同 V10.4 逐字一樣，零行為改動）
    //   均價模式　：① 開咗「★ 用手柄接手」且手柄有值 → 用手柄（絕對值，零偏移）
    //               ② 否則 → 均價起步位 + 整體上下移（maOff）
    //   ★ V10.8：接手改由【明確開關】maUseHandle 話事（預設關）。
    //     舊版 botHandleOK 用 `botSet and (not botForeign)`，而 botSet 只睇「係唔係 0」
    //     → 手柄只要唔係 0 就即刻接手，包括【上隻股票殘留嘅舊值】
    //     → 底邊 380（388 殘留）、頂邊 60（均價 +20%）→ 380 > 60 ← 用戶實測問題。
    bool botHandleOK = botSet and (not maForce) and maUseHandle
    bool topHandleOK = topSet and (not maForce) and maUseHandle
    float hBotE = useMA ? (botHandleOK ? hBot : (maBotP + maOff)) : hBot
    float hTopE = useMA ? (topHandleOK ? hTop : (maTopP + maOff)) : hTop
    // ★ V10.8：接手值離均價位好遠 → 好可能係上隻股票殘留（純提示，唔阻止）
    float maBand = math.abs(maTopP - maBotP)
    float maFarTol = math.max(maBand * 3.0, maPrice * 0.5)
    bool botFar = useMA and maUseHandle and botSet and (math.abs(hBot - maBotP) > maFarTol)
    bool topFar = useMA and maUseHandle and topSet and (math.abs(hTop - maTopP) > maFarTol)

    // 有效手柄（未被適應）—— 繪製一律用呢兩個
    // ★ V10.7：均價模式【強制 true】—— 否則會被 adapted 蓋過，
    //   方框會跌返「現價自動位置」，均價起步位就永遠見唔到。
    bool botOnEff = useMA ? true : (botOn and not adapted)
    bool topOnEff = useMA ? true : (topOn and not adapted)

    // ═══ 組裝：手柄 / 均價起步 > 自動位置 ═══
    float yBotSet = botOnEff ? hBotE : (yAutoBot + offY)
    float yTopSet = topOnEff ? hTopE : (yBotSet + refH)

    // ★ V10.10 精簡：⑩ 安全網 同 ⑩ 垂直鎖定 已移除 →
    //   唔再需要 pullBack / hKeep 兩層「搬走方框」嘅中間變數。
    //   ★ v9.2 原則保留：兩條邊交叉時唔回退高度 —— 用 min/max，
    //     方框永遠由低至高，而兩條邊【依然各自貼實自己嘅手柄】。
    float yBot = math.min(yBotSet, yTopSet)
    float yTop = math.max(yBotSet, yTopSet)
    float boxH = math.max(yTop - yBot, 1e-9)
    float cellH = boxH / rows

    // ── 平鋪：第 (mr, k) 格 = 起點 + k·格寬 + mr·格高 → 零間隙首尾相連 ──
    for mr = 0 to rows - 1
        for k = 0 to cols - 1
            int idx = mr * cols + k
            if idx < 5
                // ★ v9.3：⑥ 主開關關閉時，sepOn = 0 → 所有單格位移一律失效
                float sepOn = sepEnable ? 1.0 : 0.0
                float cOffX = (sep1X * (idx == 0 ? 1.0 : 0.0) + sep2X * (idx == 1 ? 1.0 : 0.0) + sep3X * (idx == 2 ? 1.0 : 0.0) + sep4X * (idx == 3 ? 1.0 : 0.0) + sep5X * (idx == 4 ? 1.0 : 0.0)) * sepOn
                float cOffY = (sep1Y * (idx == 0 ? 1.0 : 0.0) + sep2Y * (idx == 1 ? 1.0 : 0.0) + sep3Y * (idx == 2 ? 1.0 : 0.0) + sep4Y * (idx == 3 ? 1.0 : 0.0) + sep5Y * (idx == 4 ? 1.0 : 0.0)) * sepOn
                float cx = xLeftBar + k * cellWBars + cellWBars * cOffX / 100.0
                float cy = yBot + mr * cellH + cellH * cOffY / 100.0
                f_box(cx, cy, cellWBars, cellH, xUseBar)

    // ── 結算日垂直參考線 ──
    float sTop = yTop + cellH * 0.35
    float sBot = yBot - cellH * 0.25
    // ★ V10.0：結算日線改用【同方框邊線完全一樣嘅座標路徑】。
    //   舊版：結算日線用 xloc.bar_time、方框用 xloc.bar_index → 兩套座標系
    //   → 未來結算日時兩者會分開（正是「右邊唔貼結算日」）。
    //   新版：一律用 f_tsToBar(snapTs) 嘅同一個 K 棒索引 + xloc.bar_index
    //   → 數學上必定重疊。
    float settleBar = f_tsToBar(float(snapTs))
    if showSettleLine
        if xUseBar
            f_lnS(settleBar, sBot, settleBar, sTop, cSettle, 1, line.style_dashed)
        else
            f_lnT(float(snapTs), sBot, float(snapTs), sTop, cSettle, 1, line.style_dashed)
        if settleLabel
            string qTxt = str.format_time(snapTs, "yyyy-MM-dd") + "  " + (eMM <= 3 ? "Q1" : eMM <= 6 ? "Q2" : eMM <= 9 ? "Q3" : "Q4")
            if xUseBar
                f_qlabB(settleBar, sTop + cellH * 0.02, qTxt)
            else
                f_qlab(float(snapTs), sTop + cellH * 0.02, qTxt)

    // ── 原始大小虛線框 ──
    if showRef
        f_ref(xLeftBar, yBot, baseBars, baseH, xUseBar)

    // ★ V10.10 精簡：⑪ 對位診斷（螢光位置線 diagMode）已移除。
    // ═══ ★ 四邊手柄指示標籤 ═══
    //   注意：呢啲標籤只係「話你知邊條邊由邊個手柄控制」，
    //   真正要拖嘅係圖上嗰四個【圓點標記】，標籤本身係拖唔到嘅。
    if showHandle
        // ★ v9.2：手柄 4 要講清楚「右邊跟結算日」勾選時會暫停。
        // ★ V10.10：lockTag（垂直鎖定）／slotTag（記憶表）已移除。
        string rsTag = rightFollowSettle ? "　[跟結算日・唔跟手柄]" : ""
        // ★ v9.4：適應觸發時，手柄標記本身喺畫面外 → 標籤要講清楚，
        //   否則又會出現「拖咗但方框冇反應」嘅誤解（v8.3 教訓）。
        string adaptTag = adapted ? "　[已適應・原手柄唔屬當前標的]" : ""
        // ★ V10.1：明確指出「可拖曳標記喺畫面外 → 見唔到、拖唔到」
        //   根因：input.time() / input.price() 嘅標記由 TradingView 按數值擺位，
        //         數值唔喺可見範圍 → 標記完全消失。
        //   ★ 唔講清楚嘅話，用戶會以為「手柄壞咗」（v8.3 教訓）。
        // ★ V10.7：均價模式嘅三種狀態都要講清楚（接手 / 重設中 / 未設定）
        // ★ V10.8：均價模式三種狀態（已接手 / 重設中 / 未接手 → 用滑桿）
        string maTag1 = useMA ? (botHandleOK ? "　[★ 均價起步 → 已由手柄接手]" : maForce ? "　[★ 已按「重設返均價起步」→ 暫時唔理手柄]" : maUseHandle ? "　[★ 想拖：狀態欄有建議值]" : "　[★ 均價起步・手柄未接手（用「整體上下移」滑桿）]") : ""
        string maTag3 = useMA ? (topHandleOK ? "　[★ 均價起步 → 已由手柄接手]" : maForce ? "　[★ 已按「重設返均價起步」→ 暫時唔理手柄]" : maUseHandle ? "　[★ 想拖：狀態欄有建議值]" : "　[★ 均價起步・手柄未接手（用「整體上下移」滑桿）]") : ""
        string offTag1 = useMA ? maTag1 : botForeign ? "　[⚠ 已適應・圓點喺畫面外・拖唔到]" : botOn ? "" : "　[⚠ 未設定價格 → 方框自動擺位]"
        string offTag2 = leftOn ? "" : "　[⚠ 日期喺畫面外・拖唔到 → 請向左滾動]"
        string offTag3 = useMA ? maTag3 : topForeign ? "　[⚠ 已適應・圓點喺畫面外・拖唔到]" : topOn ? "" : "　[⚠ 未設定價格 → 方框自動擺位]"
        string offTag4 = (rightFollowSettle or rightOn) ? "" : "　[⚠ 日期喺畫面外・拖唔到 → 請向右滾動]"
        f_handle(xLeftBar + boxWBars / 2, yBot + cellH * 0.30, "▲ 手柄2 → 底邊（上下拖）" + adaptTag + offTag1, cH2, xUseBar)
        f_handle(xLeftBar, yBot + boxH / 2, "◀ 手柄3 → 左邊（左右拖）" + offTag2, cH3, xUseBar)
        f_handle(xLeftBar + boxWBars / 2, yTop - cellH * 0.30, "▼ 手柄1 → 頂邊（上下拖）" + adaptTag + offTag3, cH1, xUseBar)
        f_handle(xRightBar, yBot + boxH * 0.25, "▶ 手柄4 → 右邊（左右拖）" + rsTag + offTag4, cH4, xUseBar)

    // ═══ 狀態標籤 ═══
    if showInfo
        // ★ V10.10：記憶表已移除 → 手柄狀態只剩「均價起步 / 已設定 / 未設定」。
        // ★ V10.4：未設定（=0）時亦要顯示【建議值】—— 否則用戶唔知要打咩數先拖得到。
        // ★ V10.7：均價模式加分支 —— 講清楚「起步位 / 已接手 / 重設中」三種狀態
        string hTxt1 = useMA ? (botHandleOK ? "★ 均價起步 → 手柄已接手 " + str.tostring(hBot, "#.####") + "（起步位 " + str.tostring(maBotP + maOff, "#.####") + "）" + (botFar ? "　⚠ 離均價位好遠（可能係上隻股票殘留）" : "") : maForce ? "★ 均價起步 " + str.tostring(maBotP + maOff, "#.####") + "（已按重設・暫時唔理手柄 " + str.tostring(hBot, "#.####") + "）" : maUseHandle ? "★ 均價起步 " + str.tostring(maBotP + maOff, "#.####") + "　★ 想拖：喺設定「★ 底邊價格」打 " + str.tostring(yBot, "#.####") : "★ 均價起步 " + str.tostring(maBotP + maOff, "#.####") + "（手柄未接手・用「整體上下移」滑桿調）　★ 想拖圓點：先開「★ 用手柄接手」再喺「★ 底邊價格」打 " + str.tostring(yBot, "#.####")) : not botOn ? "○ 未設定（=0）→ 方框自動擺位　★ 想拖：喺設定打 " + str.tostring(yBot, "#.####") : botOnEff ? "✓ 已設定 " + str.tostring(hBot, "#.####") : "⚠ 原值 " + str.tostring(hBot, "#.####") + " 唔屬當前標的 → 建議改為 " + str.tostring(yBot, "#.####")
        // ★ V10.1：唔可以再一律寫「未設定」—— 明明有設定，只係唔喺畫面內。
        string hTxt2 = leftOn ? "✓ 已設定 " + str.format_time(int(hLeft), "yyyy-MM-dd") + "（↔ 邊線差 " + str.tostring(xLeftBar - f_tsToBar(float(hLeft)), "#.##") + " 棒）" : "⚠ 日期 " + str.format_time(int(hLeft), "yyyy-MM-dd") + " 喺有效範圍外 → 左邊改用自動推算"
        string hTxt3 = useMA ? (topHandleOK ? "★ 均價起步 → 手柄已接手 " + str.tostring(hTop, "#.####") + "（起步位 " + str.tostring(maTopP + maOff, "#.####") + "）" + (topFar ? "　⚠ 離均價位好遠（可能係上隻股票殘留）" : "") : maForce ? "★ 均價起步 " + str.tostring(maTopP + maOff, "#.####") + "（已按重設・暫時唔理手柄 " + str.tostring(hTop, "#.####") + "）" : maUseHandle ?
