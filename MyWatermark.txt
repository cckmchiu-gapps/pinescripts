string  i_text1     = input("Guidelines :-                            \n" +
     "1. Identify trend and structure\n" +
     "2. Prepare and Predict            \n" +
     "3. Take Action and Enter         \n" +
     "4. 心態穩定，風險管理           ", "Text 1", inline = "11")
string  i_text2     = input("Eric CHIU", "Text 2", inline = "11", tooltip = "Clear 'Text 2' to prevent animation.")
string  i_tableYpos = input("top", "Position", inline = "12", options = ["top", "middle", "bottom"])
string  i_tableXpos = input("left", "", inline = "12", options = ["left", "center", "right"])
int     i_height    = input(25, "Height", minval = 1, maxval = 100, inline = "13")
int     i_width     = input(170, "Width",  minval = 1, maxval = 100, inline = "13", tooltip = "1-100")
color   i_c_text    = input(color.new(color.black, 0), "Text", inline = "14")
string  i_textSize  = input("large", "Size", inline = "44", options = ["tiny", "small", "normal", "large", "huge", "auto"])
color   i_c_bg      = input(color.new(color.white, 10), "Background")

// We use `var` to only initialize the table on the first bar.
var table watermark = table.new(i_tableYpos + "_" + i_tableXpos, 1, 1)

// We only populate the table on the last bar; it's more efficient.
if barstate.islast
    // This `varip` variable will preserve its value across realtime updates.
    varip bool _changeText = true
    // Toggle this value on each update.
    _changeText := not _changeText
    // If there's a "Text 2" string in inputs and it's time to flip, change the text.
    string _txt = str.length(i_text2) != 0 and _changeText ? i_text2 : i_text1
    // Populate our table cell.
    table.cell(watermark, 0, 0, _txt, i_width, i_height, i_c_text, text_size = i_textSize, bgcolor = i_c_bg)
